#include "raylib.h"
#include <stdlib.h>
#include <string.h>

#define COLS 10
#define ROWS 20
#define CELL 32
#define PANEL_W 160

#define SCREEN_W (COLS * CELL + PANEL_W)
#define SCREEN_H (ROWS * CELL)

typedef struct {
    int x, y;
} Vec2;

typedef struct {
    Vec2 cells[4];
    Color color;
} Piece;

// 7 tetrominos (SRS standard)
static const Vec2 SHAPES[7][4] = {
    {{0,1},{1,1},{2,1},{3,1}},  // I
    {{0,0},{0,1},{1,1},{2,1}},  // J
    {{2,0},{0,1},{1,1},{2,1}},  // L
    {{1,0},{2,0},{1,1},{2,1}},  // S  (was O, now corrected)
    {{0,0},{1,0},{1,1},{2,1}},  // Z  (was S)
    {{1,0},{0,1},{1,1},{2,1}},  // T
    {{0,0},{1,0},{2,0},{1,1}},  // O-like / extra
};

// Fix: proper 7 tetrominoes
static const Vec2 TETROMINOS[7][4] = {
    {{0,1},{1,1},{2,1},{3,1}},  // I
    {{0,0},{0,1},{1,1},{2,1}},  // J
    {{2,0},{0,1},{1,1},{2,1}},  // L
    {{1,0},{2,0},{1,1},{2,1}},  // S
    {{0,0},{1,0},{1,1},{2,1}},  // Z
    {{1,0},{0,1},{1,1},{2,1}},  // T
    {{1,0},{2,0},{1,1},{2,1}},  // O  -- actually S; real O below
};

static const Color COLORS[7] = {
    {0, 240, 240, 255},   // I - cyan
    {0, 0, 240, 255},     // J - blue
    {240, 160, 0, 255},   // L - orange
    {240, 240, 0, 255},   // S - yellow
    {0, 240, 0, 255},     // Z - green
    {160, 0, 240, 255},   // T - purple
    {240, 0, 0, 255},     // O - red
};

// Correct standard tetrominos
static const Vec2 TET[7][4] = {
    {{0,1},{1,1},{2,1},{3,1}},  // I
    {{0,0},{0,1},{1,1},{2,1}},  // J
    {{2,0},{0,1},{1,1},{2,1}},  // L
    {{1,0},{2,0},{0,1},{1,1}},  // S
    {{0,0},{1,0},{1,1},{2,1}},  // Z
    {{1,0},{0,1},{1,1},{2,1}},  // T
    {{1,0},{2,0},{1,1},{2,1}},  // O (square: offset)
};

typedef struct {
    Vec2 pos;
    Vec2 cells[4];
    Color color;
    int type;
} ActivePiece;

int board[ROWS][COLS];       // 0 = empty, 1..7 = color index
Color boardColor[ROWS][COLS];

int score = 0;
int level = 1;
int lines = 0;
bool gameOver = false;
bool paused = false;

ActivePiece current, next;

void SpawnPiece(ActivePiece *p, int type) {
    p->type = type;
    p->color = COLORS[type];
    for (int i = 0; i < 4; i++) p->cells[i] = TET[type][i];
    p->pos.x = COLS / 2 - 2;
    p->pos.y = 0;
}

int RandomType(void) {
    return GetRandomValue(0, 6);
}

bool IsValid(ActivePiece *p, int dx, int dy) {
    for (int i = 0; i < 4; i++) {
        int nx = p->pos.x + p->cells[i].x + dx;
        int ny = p->pos.y + p->cells[i].y + dy;
        if (nx < 0 || nx >= COLS || ny >= ROWS) return false;
        if (ny >= 0 && board[ny][nx]) return false;
    }
    return true;
}

void RotatePiece(ActivePiece *p) {
    // I piece special pivot
    Vec2 tmp[4];
    for (int i = 0; i < 4; i++) {
        tmp[i].x =  p->cells[i].y;
        tmp[i].y = -p->cells[i].x + 3;
    }
    // Normalize so min x/y == 0
    int minX = 99, minY = 99;
    for (int i = 0; i < 4; i++) {
        if (tmp[i].x < minX) minX = tmp[i].x;
        if (tmp[i].y < minY) minY = tmp[i].y;
    }
    for (int i = 0; i < 4; i++) { tmp[i].x -= minX; tmp[i].y -= minY; }

    ActivePiece test = *p;
    for (int i = 0; i < 4; i++) test.cells[i] = tmp[i];
    // Wall kick: try offsets 0, -1, +1
    int kicks[] = {0, -1, 1, -2, 2};
    for (int k = 0; k < 5; k++) {
        if (IsValid(&test, kicks[k], 0)) {
            test.pos.x += kicks[k];
            *p = test;
            return;
        }
    }
}

void LockPiece(ActivePiece *p) {
    for (int i = 0; i < 4; i++) {
        int x = p->pos.x + p->cells[i].x;
        int y = p->pos.y + p->cells[i].y;
        if (y >= 0 && y < ROWS && x >= 0 && x < COLS) {
            board[y][x] = p->type + 1;
            boardColor[y][x] = p->color;
        }
    }
}

int ClearLines(void) {
    int cleared = 0;
    for (int y = ROWS - 1; y >= 0; y--) {
        bool full = true;
        for (int x = 0; x < COLS; x++) {
            if (!board[y][x]) { full = false; break; }
        }
        if (full) {
            cleared++;
            for (int row = y; row > 0; row--) {
                memcpy(board[row], board[row-1], sizeof(board[0]));
                memcpy(boardColor[row], boardColor[row-1], sizeof(boardColor[0]));
            }
            memset(board[0], 0, sizeof(board[0]));
            y++; // re-check same row
        }
    }
    return cleared;
}

void ResetGame(void) {
    memset(board, 0, sizeof(board));
    memset(boardColor, 0, sizeof(boardColor));
    score = 0; level = 1; lines = 0;
    gameOver = false; paused = false;
    SpawnPiece(&next, RandomType());
    SpawnPiece(&current, RandomType());
    next = (ActivePiece){0};
    SpawnPiece(&next, RandomType());
}

int main(void) {
    InitWindow(SCREEN_W, SCREEN_H, "TETRIS - C + Raylib");
    SetTargetFPS(60);

    ResetGame();

    float fallTimer = 0.0f;
    float lockTimer = 0.0f;
    bool landing = false;
    float moveRepeat = 0.0f;
    int moveDir = 0;

    while (!WindowShouldClose()) {
        float dt = GetFrameTime();

        if (!gameOver && !paused) {
            // --- Input ---
            if (IsKeyPressed(KEY_UP))    RotatePiece(&current);
            if (IsKeyPressed(KEY_SPACE)) {
                while (IsValid(&current, 0, 1)) current.pos.y++;
                landing = true; lockTimer = 0.3f;
            }
            if (IsKeyPressed(KEY_LEFT))  { moveDir = -1; moveRepeat = 0.2f; if (IsValid(&current,-1,0)) current.pos.x--; }
            if (IsKeyPressed(KEY_RIGHT)) { moveDir =  1; moveRepeat = 0.2f; if (IsValid(&current, 1,0)) current.pos.x++; }
            if (IsKeyReleased(KEY_LEFT) || IsKeyReleased(KEY_RIGHT)) moveDir = 0;

            // Auto-repeat move
            if (moveDir != 0) {
                moveRepeat -= dt;
                if (moveRepeat <= 0) {
                    moveRepeat = 0.05f;
                    if (IsValid(&current, moveDir, 0)) current.pos.x += moveDir;
                }
            }

            // Soft drop
            float speed = (level <= 10) ? (0.8f - (level-1)*0.07f) : 0.1f;
            if (IsKeyDown(KEY_DOWN)) speed = 0.05f;

            fallTimer += dt;
            if (fallTimer >= speed) {
                fallTimer = 0;
                if (IsValid(&current, 0, 1)) {
                    current.pos.y++;
                    landing = false;
                } else {
                    landing = true;
                }
            }

            // Lock delay
            if (landing) {
                lockTimer += dt;
                if (lockTimer >= 0.4f) {
                    LockPiece(&current);
                    int cleared = ClearLines();
                    lines += cleared;
                    int pts[] = {0, 100, 300, 500, 800};
                    if (cleared > 0) score += pts[cleared] * level;
                    level = lines / 10 + 1;

                    current = next;
                    SpawnPiece(&next, RandomType());

                    if (!IsValid(&current, 0, 0)) gameOver = true;
                    landing = false; lockTimer = 0;
                }
            }
        }

        if (IsKeyPressed(KEY_P)) paused = !paused;
        if (gameOver && IsKeyPressed(KEY_R)) ResetGame();

        // --- Ghost piece ---
        ActivePiece ghost = current;
        while (IsValid(&ghost, 0, 1)) ghost.pos.y++;

        // --- Draw ---
        BeginDrawing();
        ClearBackground((Color){18, 18, 30, 255});

        // Grid background
        for (int y = 0; y < ROWS; y++)
            for (int x = 0; x < COLS; x++)
                DrawRectangleLines(x*CELL, y*CELL, CELL, CELL, (Color){40,40,60,255});

        // Board cells
        for (int y = 0; y < ROWS; y++)
            for (int x = 0; x < COLS; x++)
                if (board[y][x]) {
                    DrawRectangle(x*CELL+1, y*CELL+1, CELL-2, CELL-2, boardColor[y][x]);
                    DrawRectangle(x*CELL+1, y*CELL+1, CELL-2, 4, Fade(WHITE, 0.3f));
                }

        // Ghost
        for (int i = 0; i < 4; i++) {
            int gx = (ghost.pos.x + ghost.cells[i].x) * CELL;
            int gy = (ghost.pos.y + ghost.cells[i].y) * CELL;
            DrawRectangle(gx+1, gy+1, CELL-2, CELL-2, Fade(ghost.color, 0.2f));
            DrawRectangleLines(gx+1, gy+1, CELL-2, CELL-2, Fade(ghost.color, 0.5f));
        }

        // Current piece
        if (!gameOver) {
            for (int i = 0; i < 4; i++) {
                int cx = (current.pos.x + current.cells[i].x) * CELL;
                int cy = (current.pos.y + current.cells[i].y) * CELL;
                DrawRectangle(cx+1, cy+1, CELL-2, CELL-2, current.color);
                DrawRectangle(cx+1, cy+1, CELL-2, 4, Fade(WHITE, 0.3f));
            }
        }

        // Side panel
        int px = COLS * CELL + 10;
        DrawText("TETRIS", px, 20, 22, WHITE);
        DrawLine(px, 50, SCREEN_W - 10, 50, (Color){80,80,120,255});

        DrawText("SCORE", px, 60, 14, GRAY);
        DrawText(TextFormat("%d", score), px, 78, 20, WHITE);

        DrawText("LEVEL", px, 110, 14, GRAY);
        DrawText(TextFormat("%d", level), px, 128, 20, WHITE);

        DrawText("LINES", px, 160, 14, GRAY);
        DrawText(TextFormat("%d", lines), px, 178, 20, WHITE);

        // Next piece preview
        DrawText("NEXT", px, 215, 14, GRAY);
        DrawRectangleLines(px, 232, 5*CELL/2, 5*CELL/2, (Color){80,80,120,255});
        for (int i = 0; i < 4; i++) {
            int nx2 = px + next.cells[i].x * (CELL-4) + 4;
            int ny2 = 236 + next.cells[i].y * (CELL-4) + 4;
            DrawRectangle(nx2, ny2, CELL-6, CELL-6, next.color);
        }

        // Controls hint
        DrawText("[←→] Move",  px, SCREEN_H-120, 11, DARKGRAY);
        DrawText("[↑]  Rotate", px, SCREEN_H-104, 11, DARKGRAY);
        DrawText("[↓]  Soft drop", px, SCREEN_H-88, 11, DARKGRAY);
        DrawText("[SPC] Hard drop", px, SCREEN_H-72, 11, DARKGRAY);
        DrawText("[P]  Pause",  px, SCREEN_H-56, 11, DARKGRAY);

        if (paused) {
            DrawRectangle(0, 0, COLS*CELL, ROWS*CELL, Fade(BLACK, 0.6f));
            DrawText("PAUSED", COLS*CELL/2 - 50, ROWS*CELL/2 - 12, 30, WHITE);
        }

        if (gameOver) {
            DrawRectangle(0, 0, COLS*CELL, ROWS*CELL, Fade(BLACK, 0.7f));
            DrawText("GAME OVER", COLS*CELL/2 - 75, ROWS*CELL/2 - 30, 28, RED);
            DrawText(TextFormat("Score: %d", score), COLS*CELL/2 - 50, ROWS*CELL/2 + 10, 20, WHITE);
            DrawText("[R] Restart", COLS*CELL/2 - 50, ROWS*CELL/2 + 40, 18, YELLOW);
        }

        EndDrawing();
    }

    CloseWindow();
    return 0;
}
